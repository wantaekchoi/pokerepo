<!-- POKEREPO:START -->
<!-- POKEREPO:END -->

같은 저장소에 .github/workflows/pokerepo.yml 파일을 만들고 아래를 붙인다. 그다음 Actions 탭에서 한 번 실행하면, 이후로는 매일 알아서 돈다.

name: pokerepo
on:
  schedule: [{ cron: "0 21 * * *" }]
  workflow_dispatch:
permissions: { contents: write }
jobs:
  run:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: wantaekchoi/pokerepo@v1
